# etf 抄底研究：把每日收盘价和资金流拼到一起
engine_ts = create_engine('mysql+pymysql://user:pw@host/quant')


def list_etf(symbols, days=60):
    # 返回最近 N 天的收盘价
    sql = """SELECT * FROM etf_basic WHERE code IN ({})"""
    df = pd.read_sql_query(sql, engine_ts)
    if df.empty:
        raise ValueError(f"no rows for {symbols}")
    return df.tail(days)
